def download_street_network_from_gdf(
    gdf:gpd.GeoDataFrame,
    save_path:str,
):
    print(f"Gathering street network for {save_path}")
    if not os.path.exists(save_path):
        gdf_copy = gdf.copy()
        gdf_copy = gdf_copy.to_crs(4326)
        single_hull = gdf_copy.union_all().convex_hull
        street_network = ox.graph_from_polygon(single_hull, network_type = "drive")
    
        street_network_gdf = ox.graph_to_gdfs(street_network, nodes = False)
        street_network_gdf.to_file(save_path)

Segmenting blocks with momepy

Now I’ll use momepy’s enclosed_tesellation function to obtain the blocks using the street network enclosures.

import momepy


def export_blocks(
    street_network_gdf, 
    save_path_blocks,
):
    if not os.path.exists(save_path_blocks):
        street_network_gdf = street_network_gdf.to_crs(street_network_gdf.estimate_utm_crs())
        blocks = momepy.enclosures(street_network_gdf)
        blocks.to_file(save_path_blocks)
    
    blocks = gpd.read_file(save_path_blocks)

    return blocks

def get_window_center(
    gdf_footprints,
    crs,
    center_lon_lat = None,
):
    # Default window center: the median building, which lands in the built-up area
    if center_lon_lat is None:
        centroids = gdf_footprints.to_crs(crs).geometry.centroid
        return centroids.x.median(), centroids.y.median()

    center = gpd.GeoSeries(gpd.points_from_xy([center_lon_lat[0]], [center_lon_lat[1]]), crs = 4326).to_crs(crs)
    return center.x[0], center.y[0]


def plot_window_contours(gdf, center, color, title):
    # The basemap is in web mercator (EPSG:3857), so the window is converted to it
    center_x, center_y = center
    window = box(center_x - HALF_WINDOW, center_y - HALF_WINDOW, center_x + HALF_WINDOW, center_y + HALF_WINDOW)
    x_min, y_min, x_max, y_max = gpd.GeoSeries([window], crs = gdf.crs).to_crs(3857).total_bounds
    gdf_window = gdf.to_crs(3857).cx[x_min:x_max, y_min:y_max]

    map_visualizer = MapVisualizer()
    map_visualizer.initialize_map(figsize = (6, 6))
    # Contours only, so the basemap stays visible
    gdf_window.boundary.plot(ax = map_visualizer.ax, color = color, linewidth = 1.5)
    map_visualizer.ax.set_xlim(x_min, x_max)
    map_visualizer.ax.set_ylim(y_min, y_max)
    map_visualizer.add_basemap(alpha = 0.8, api_key = CARTO_API_KEY)
    map_visualizer.finalize_map(title = title)

# One window takes about 15 s and 5 GB of RAM.
D0 = 9.0


def get_marching_squares_blocks(
    gdf_footprints,
    crs,
    save_path,
    center_lon_lat = None,
):
    if not os.path.exists(save_path):
        gdf_metric = gdf_footprints.to_crs(crs)
        center_x, center_y = get_window_center(gdf_metric, crs, center_lon_lat)

        gdf_window = gdf_metric.cx[
            center_x - HALF_WINDOW:center_x + HALF_WINDOW,
            center_y - HALF_WINDOW:center_y + HALF_WINDOW,
        ]

        distance_field, transform = get_distance_to_footprint_field(gdf_window)

        marching_squares = MarchingSquares(distance_field)
        marching_squares.march(D0)
        polygons = marching_squares.to_polygons(transform, crs)
        polygons.to_file(save_path)

    polygons = gpd.read_file(save_path)

    return polygons

Our last step is running the actual comparisons and plotting. The metrics being displayed in the following plot are:

  1. Segmentation quality (SQ): Keeping the pairs of blocks for which \(IoU > m = 0.5\), compute the mean IoU. Since we are imposing a threshold \(m\) the average will be higher than \(m\), so we can expect scores above \(0.5\) for this metric.

  2. Recognition quality (RQ): F-Score of the matched blocks, taking the momepy blocks as the ground truth.

  3. Panoramic Quality (PQ): It’s defined as the product of \(SQ\) and \(RQ\):

\[ PQ = RQ \times SQ \]

  1. Boundary F at distinct radiuses. How much do the boundaries of the marching squares and momepy blocks intersects, using an allowance buffer of \(2\)~m, \(5\)~m and \(10\)~m.
from shapely.geometry import box
from footprint_metrics import get_average_iou, get_panoptic_quality, get_boundary_f_score
from imageable.visualization.data_visualization import plot_grouped_distribution_bars

BOUNDARY_TOLERANCES = [2.0, 5.0, 10.0]


def get_evaluation_layers(
    gdf_blocks,
    gdf_polygons,
    gdf_footprints,
):
    # Clip the enclosures to the marching squares window, so both methods are cut at the same edge.
    # Keep the enclosures that contain at least one building.
    window = box(*gdf_polygons.total_bounds)
    gdf_blocks = gdf_blocks.clip(window, keep_geom_type = True)
    gdf_footprints_window = gdf_footprints.to_crs(gdf_blocks.crs).clip(window)
    blocks_with_buildings = gpd.sjoin(gdf_blocks, gdf_footprints_window[["geometry"]], predicate = "intersects").index
    gdf_blocks = gdf_blocks[gdf_blocks.index.isin(blocks_with_buildings)]

    # Void rule of panoptic quality: drop the predictions that lie mostly outside the kept enclosures
    evaluation_area = gdf_blocks.union_all()
    share_inside = gdf_polygons.intersection(evaluation_area).area / gdf_polygons.area
    gdf_polygons = gdf_polygons[share_inside > 0.5]

    return gdf_blocks, gdf_polygons
